# Тестирование Этапа 4: команды ls, cd, cat, whoami
# 1. Смотрим корень
ls
# 2. Идём в папку docs
cd docs
ls
# 3. Идём глубже - в api
cd api
ls
# 4. Смотрим содержимое файла
cat reference.md
# 5. Возвращаемся на два уровня вверх
cd ..
cd ..
ls
# 6. Пробуем зайти в несуществующую папку
cd nonexistent
# 7. Пробуем cat папки (должна быть ошибка)
cat docs
# 8. Пробуем cat несуществующего файла
cat missing.txt
# 9. Имя пользователя
whoami
# 10. Идём в src и смотрим вложенность
cd src
ls
cd utils/data
ls
cat values.txt
# 11. Абсолютный путь
cd /
ls
# 12. Завершаем
exit
