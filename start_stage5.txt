# 1. Смотрим корень до изменений
ls -l
# 2. Меняем владельца файла readme.txt на 'admin'
chown admin readme.txt
# 3. Проверяем - владелец должен измениться
ls -l
# 4. Меняем владельца папки docs на 'admin'
chown admin docs
# 5. Смотрим
ls -l
# 6. Заходим внутрь docs - файлы там остались с прежним владельцем
cd docs
ls -l
# 7. Меняем владельца guide.md
chown admin guide.md
ls -l
# 8. Возвращаемся в корень
cd ..
# 9. Пробуем chown на несуществующем файле (ошибка)
chown admin missing.txt
# 10. Пробуем chown без аргументов (ошибка)
chown admin
# 11. Пробуем chown без аргументов вообще (ошибка)
chown
# 12. Проверяем, что cat всё ещё работает
cat readme.txt
# 13. Проверяем whoami
whoami
# 14. Выходим
exit
